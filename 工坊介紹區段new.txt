<!-- =====================================
     JavaScript：五張圖片輪播功能
===================================== -->

<script>
(() => {

  const section = document.currentScript.closest(
    '.workshop-section'
  );

  if (!section) return;

  const slides = [
    ...section.querySelectorAll('.ws-slide')
  ];

  const dots = [
    ...section.querySelectorAll('.ws-dot')
  ];

  const prev = section.querySelector('.ws-prev');
  const next = section.querySelector('.ws-next');
  const caption = section.querySelector('.ws-caption');

  const descriptions = [
    '三峽染工坊｜藍染文化與手作體驗',
    '三峽染工坊｜傳統染布工藝',
    '三峽染工坊｜藍染手作成果',
    '三峽染工坊｜天然植物染色',
    '三峽染工坊｜特色商品展售'
  ];

  let current = 2;

  function updateCarousel() {

    slides.forEach((slide, index) => {

      const position = (index - current + 5) % 5;

      slide.dataset.position = position;

    });

    dots.forEach((dot, index) => {

      dot.classList.toggle(
        'active',
        index === current
      );

    });

    caption.textContent = descriptions[current];

  }

  function nextSlide() {

    current = (current + 1) % 5;
    updateCarousel();

  }

  function prevSlide() {

    current = (current - 1 + 5) % 5;
    updateCarousel();

  }

  next.addEventListener('click', nextSlide);
  prev.addEventListener('click', prevSlide);

  dots.forEach((dot, index) => {

    dot.addEventListener('click', () => {

      current = index;
      updateCarousel();

    });

  });

  // 支援觸控滑動
  let startX = 0;

  const carousel = section.querySelector('.ws-carousel');

  carousel.addEventListener('touchstart', (event) => {

    startX = event.changedTouches[0].clientX;

  }, { passive: true });

  carousel.addEventListener('touchend', (event) => {

    const endX = event.changedTouches[0].clientX;
    const distance = endX - startX;

    if (Math.abs(distance) > 45) {

      if (distance < 0) {
        nextSlide();
      } else {
        prevSlide();
      }

    }

  }, { passive: true });

  updateCarousel();

})();
</script>

</section>
